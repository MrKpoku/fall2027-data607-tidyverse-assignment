CONVERTING THE COLUMNS- PIVOT LONGER

Code 
## I used pivot longer to reshape the data from wide to long format.it combines the four card network transaction column and identifies each card network.

card_association_2 <- card_association |>
  pivot_longer(
    cols = c(
      visa_transactions_millions,
      mastercard_transactions_millions,
      amex_transactions_millions,
      discover_transactions_millions
    ),
    names_to = "network",
    names_pattern = "(.*)_transactions_millions",
    values_to = "transactions_millions"
  ) |>
  mutate(
    network = recode(
      network,
      visa = "Visa",
      mastercard = "Mastercard",
      amex = "American Express",
      discover = "Discover"
    )
  ) |>
  select(
    month,
    year,
    network,
    transactions_millions
  )

EXPLORATORY ANALYSIS - NETWORK VS TRANSACTIONS

Code 
## here i wanted to discover which network has the highest transactions and % contribution.it concluded with american express being the highest

network_totals <- card_association_2 |>
  group_by(network) |>
  summarise(
    transactions_millions = sum(transactions_millions, na.rm = TRUE),
    .groups = "drop"
  ) |>
  mutate(
    percent_contribution =
      round(transactions_millions / sum(transactions_millions) * 100, 2),
    transactions_millions = comma(transactions_millions, accuracy = 1)
  ) |>
  arrange(desc(as.numeric(gsub(",", "", transactions_millions))))

VISUALIZATION - NETWORK VS TRANSACTIONS

Code 
## I visualized the data to make the results easier to understand, especially for readers who prefer charts to tables of numbers and code.i must say its a close distribution for the highest to the lowest

network_totals <- card_association_2 |>
  group_by(network) |>
  summarise(
    transactions_millions = sum(transactions_millions, na.rm = TRUE),
    .groups = "drop"
  ) |>
  mutate(
    percent_contribution =
      transactions_millions / sum(transactions_millions) * 100
  ) |>
  arrange(desc(transactions_millions))

ggplot(
  network_totals,
  aes(
    x = reorder(network, transactions_millions),
    y = transactions_millions,
    fill = network
  )
) +
  geom_col(show.legend = FALSE) +
  geom_text(
    aes(
      label = paste0(
        comma(transactions_millions, accuracy = 1),
        " (", round(percent_contribution, 2), "%)"
      )
    ),
    hjust = -0.1,
    size = 3.5
  ) +
  coord_flip() +
  scale_y_continuous(
    labels = comma,
    expand = expansion(mult = c(0, 0.2))
  ) +
  labs(
    title = "Total Transactions by Card Network",
    x = "Card network",
    y = "Transactions (millions)"
  ) +
  theme_minimal()
